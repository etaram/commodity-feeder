    <script>

        document.addEventListener("DOMContentLoaded", function () {

            const slider =
                document.getElementById("testiSlider");

            if (!slider) return;

            let isDown = false;
            let startX;
            let scrollLeft;
            let autoScroll = null;
            let resumeTimer = null;

            // AUTO SCROLL helpers
            function startAuto() {
                if (autoScroll) return;
                autoScroll = setInterval(() => {
                    slider.scrollLeft += 1;
                    if (slider.scrollLeft + slider.clientWidth >= slider.scrollWidth - 1) {
                        slider.scrollLeft = 0;
                    }
                }, 20);
            }

            function stopAuto() {
                clearInterval(autoScroll);
                autoScroll = null;
                clearTimeout(resumeTimer);
            }

            // Resume a little after the user stops interacting
            function resumeAutoSoon() {
                clearTimeout(resumeTimer);
                resumeTimer = setTimeout(startAuto, 2000);
            }

            // MOUSE DRAG (desktop)
            slider.addEventListener("mousedown", (e) => {
                isDown = true;
                slider.style.cursor = "grabbing";
                startX = e.pageX - slider.offsetLeft;
                scrollLeft = slider.scrollLeft;
                stopAuto();
            });

            slider.addEventListener("mousemove", (e) => {
                if (!isDown) return;
                e.preventDefault();
                const x = e.pageX - slider.offsetLeft;
                const walk = (x - startX) * 1.5;
                slider.scrollLeft = scrollLeft - walk;
            });

            slider.addEventListener("mouseup", () => {
                isDown = false;
                slider.style.cursor = "grab";
                resumeAutoSoon();
            });

            // PAUSE ON HOVER (desktop)
            slider.addEventListener("mouseenter", stopAuto);
            slider.addEventListener("mouseleave", () => {
                isDown = false;
                slider.style.cursor = "grab";
                startAuto();
            });

            // TOUCH (mobile): let native scroll move the slider, just pause auto-scroll
            slider.addEventListener("touchstart", stopAuto, { passive: true });
            slider.addEventListener("touchend", resumeAutoSoon, { passive: true });

            startAuto();

        });

                autoScroll = setInterval(() => {
                    grid.scrollLeft += 1;
                    if (grid.scrollLeft + grid.clientWidth >= grid.scrollWidth - 1) {
                        grid.scrollLeft = 0;
                    }
                }, 25);
            }

            function stopAuto() {
                clearInterval(autoScroll);
                autoScroll = null;
                clearTimeout(resumeTimer);
            }

            function resumeAutoSoon() {
                clearTimeout(resumeTimer);
                resumeTimer = setTimeout(startAuto, 2000);
            }

            // Pause while the user interacts
            grid.addEventListener("mouseenter", stopAuto);
            grid.addEventListener("mouseleave", startAuto);
            grid.addEventListener("touchstart", stopAuto, { passive: true });
            grid.addEventListener("touchend", resumeAutoSoon, { passive: true });

            // Stop/restart on resize so it only runs while horizontally scrollable
            window.addEventListener("resize", () => {
                stopAuto();
                startAuto();
            });

            startAuto();

        });


        $("#txtHomeKeywordSearch").on("keypress", function (e) {
        if (e.which === 13) {

            const value = $(this).val().trim();

            // EMPTY VALUE
            if (value === "") {

                e.preventDefault();

                return false;
            }

            // VALUE EXISTS
            $("#autoSuggestHomeForm").submit();
        }
    });
